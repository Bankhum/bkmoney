const pass = 'ใส่รหัสผ่านใหม่ที่นี่'; PropertiesService.getScriptProperties().setProperties({ ADMIN_USER: user, ADMIN_PASS: pass }); } // ============================================================ // API // ============================================================ function doGet(e) { try { checkAndInitSheet(); return jsonResponse({ success: true, status: 'online', message: 'API พร้อมใช้งานเชื่อมต่อกับ GitHub แล้ว' }); } catch (error) { return jsonResponse({ success: false, status: 'error', error: error.message }); } } function doPost(e) { try { checkAndInitSheet(); let payloadStr = ''; if (e && e.parameter && e.parameter.data) { payloadStr = e.parameter.data; } else if (e && e.postData && e.postData.contents) { payloadStr = e.postData.contents; } if (!payloadStr) throw new Error('ไม่มีข้อมูลส่งมา'); const payload = JSON.parse(payloadStr); const action = payload.action; if (!action) throw new Error('ไม่ได้ระบุคำสั่ง action'); let result; switch (action) { case 'verifyAdminLogin': result = verifyAdminLogin(payload.username, payload.password); break; case 'searchEmployee': result = searchEmployee(payload.employeeId); break; case 'getAdminDashboardData': result = getAdminDashboardData(); break; case 'saveEmployeeData': result = saveEmployeeData(payload.formData); break; case 'acknowledgeResult': result = acknowledgeResult(payload.employeeId, payload.roundName, payload.fiscalYear, payload.rating); break; case 'saveAdminSettings': result = saveAdminSettings(payload.settingsData); break; case 'getAdminSettingsPreview': result = getAdminSettingsPreview(); break; default: throw new Error('ไม่พบคำสั่งที่ระบุ (Invalid Action): ' + action); } return jsonResponse({ success: true, data: result }); } catch (error) { return jsonResponse({ success: false, error: error.message }); } } function jsonResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); if (sheet.getLastRow() === 0) { sheet.getRange(1, 1, 1, headers.length).setValues([headers]) .setBackground('#1a237e').setFontColor('#ffffff').setFontWeight('bold'); sheet.setFrozenRows(1); } let settingsSheet = ss.getSheetByName(SETTINGS_SHEET); if (!settingsSheet) { settingsSheet = ss.insertSheet(SETTINGS_SHEET); settingsSheet.getRange(1, 1, 1, 2).setValues([['Key', 'Value']]) .setBackground('#4285f4').setFontColor('#ffffff').setFontWeight('bold'); } } // ============================================================ // Login Admin (อ่านจาก Script Properties) // ============================================================ function verifyAdminLogin(username, password) { const props = PropertiesService.getScriptProperties(); const u = props.getProperty('ADMIN_USER'); const p = props.getProperty('ADMIN_PASS'); if (!u || !p) { throw new Error('ยังไม่ได้ตั้งรหัสผ่านแอดมิน (รันฟังก์ชัน setAdminCredentials ใน Apps Script ก่อน)'); } if (username === u && password === p) return true; Utilities.sleep(1500); // หน่วงเล็กน้อยลดการเดารหัส throw new Error('ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง'); } // ============================================================ // บันทึกรูปลง Google Drive // ============================================================ function saveFileToDrive(base64Data, filePrefix) { if (!base64Data || !base64Data.startsWith('data:image')) return null; if (!DRIVE_FOLDER_ID || DRIVE_FOLDER_ID === 'xxxx') { throw new Error('กรุณาตั้งค่า DRIVE_FOLDER_ID ก่อนบันทึกไฟล์ภาพ'); } const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID); const parts = base64Data.split(';base64,'); if (parts.length !== 2) throw new Error('รูปแบบข้อมูลรูปภาพไม่ถูกต้อง'); const mimeType = parts[0].replace('data:', ''); let ext = mimeType.split('/')[1] || 'png'; if (ext === 'jpeg') ext = 'jpg'; const fileName = filePrefix + '_' + new Date().getTime() + '.' + ext; const blob = Utilities.newBlob(Utilities.base64Decode(parts[1]), mimeType, fileName); const file = folder.createFile(blob); try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (error) { console.log('ไม่สามารถตั้งค่าสิทธิ์ไฟล์ได้: ' + error.message); } return file.getId(); } // ============================================================ // Settings // ============================================================ function getAdminSettings() { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SETTINGS_SHEET); if (!sheet) return { directorName: '', logoId: '', signatureId: '', stampId: '' }; const settings = {}; sheet.getDataRange().getValues().forEach(function (row) { if (row[0]) settings[row[0]] = row[1]; }); return { directorName: settings['DIRECTOR_NAME'] || '', logoId: settings['LOGO_ID'] || '', signatureId: settings['SIGNATURE_ID'] || '', stampId: settings['STAMP_ID'] || '' }; } // ส่งเป็นลิงก์รูป (เบา) แทน Base64 function driveImageUrl(fileId) { if (!fileId) return ''; sheet = ss.getSheetByName(SETTINGS_SHEET); } const updateOrInsert = function (key, value) { const data = sheet.getDataRange().getValues(); let foundRow = -1; for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(key)) { foundRow = i + 1; break; } } if (foundRow > -1) sheet.getRange(foundRow, 2).setValue(value); else sheet.appendRow([key, value]); }; if (settingsData.directorName !== undefined) { updateOrInsert('DIRECTOR_NAME', settingsData.directorName); if (id) updateOrInsert(it[2], id); } }); return 'บันทึกการตั้งค่าลงระบบเรียบร้อยแล้ว'; } // ============================================================ // ค้นหาข้อมูลบุคลากร // ============================================================ function searchEmployee(employeeId) { if (!employeeId) throw new Error('กรุณาระบุรหัสประจำตัว'); const records = rows.map(function (row) { const r = {}; headers.forEach(function (h, i) { r[h] = row[i]; }); return r; }); return { records: records, settings: getAdminSettingsPreview() }; } // ============================================================ // Dashboard // ============================================================ function getAdminDashboardData() { const sheet = SpreadsheetApp.openById(SHEET_ID).getSheetByName(SHEET_NAME); if (!sheet) throw new Error('ไม่พบชีต ' + SHEET_NAME); const rawData = sheet.getDataRange().getValues(); if (rawData.length > 0) rawData.shift(); const list = []; rawData.forEach(function (row) { if (!row[0]) return; }); newRow[16] = settings.directorName || formData.director_name || ''; let foundIndex = -1; for (let i = 1; i < data.length; i++) { if ( String(data[i][0]) === String(formData.employee_id) && String(data[i][7]) === String(formData.round_name) && String(data[i][21]) === String(formData.fiscal_year) ) { foundIndex = i + 1;
